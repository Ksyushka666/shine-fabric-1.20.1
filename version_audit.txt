--- workflow ---
name: Build Shine

on:
  push:
    branches: [ "**" ]
  pull_request:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Check out source
        uses: actions/checkout@v4
      - name: Set up Java 17
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '17'
          cache: gradle
      - name: Make Gradle executable
        run: chmod +x gradlew
      - name: Build and verify
        run: ./gradlew clean check build --no-daemon
      - name: Upload verified artifacts
        uses: actions/upload-artifact@v4
        with:
          name: shine-fabric-1.20.1-${{ github.sha }}
          path: |
            build/libs/*.jar
            PORT_STATUS.md
          if-no-files-found: error
      - name: Publish GitHub Release
        if: github.event_name == 'push' && github.ref == 'refs/heads/master'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          tag="build-${GITHUB_RUN_NUMBER}"
          gh release create "$tag" build/libs/shine-1.0.0.jar build/libs/shine-1.0.0-sources.jar \
            --repo "$GITHUB_REPOSITORY" \
            --target "$GITHUB_SHA" \
            --title "Shine Fabric 1.20.1 build ${GITHUB_RUN_NUMBER}" \
            --notes "Automated release created after a successful clean check build. Commit: ${GITHUB_SHA}." \
            --latest
